    private fun postJson(
        url: String,
        body: JSONObject
    ): JSONObject {

        val connection =
            URL(url)
                .openConnection() as HttpURLConnection

        try {

            connection.requestMethod =
                "POST"

            connection.connectTimeout =
                10_000

            connection.readTimeout =
                10_000

            connection.doOutput =
                true

            connection.setRequestProperty(
                "Content-Type",
                "application/json"
            )

            connection.setRequestProperty(
                "Accept",
                "application/json"
            )

            connection.outputStream.use { output ->

                output.write(
                    body.toString()
                        .toByteArray(
                            Charsets.UTF_8
                        )
                )
            }

            val responseCode =
                connection.responseCode

            val responseText =
                if (
                    responseCode in 200..299
                ) {

                    connection.inputStream
                        .bufferedReader()
                        .use {
                            it.readText()
                        }

                } else {

                    connection.errorStream
                        ?.bufferedReader()
                        ?.use {
                            it.readText()
                        }
                        ?: ""
                }

            if (
                responseCode !in 200..299
            ) {

                throw IllegalStateException(

                    try {

                        JSONObject(
                            responseText
                        ).optString(
                            "message",
                            "API hatasÄ±: $responseCode"
                        )

                    } catch (
                        _: Exception
                    ) {

                        "API hatasÄ±: $responseCode"
                    }
                )
            }

            return JSONObject(
                responseText
            )

        } finally {

            connection.disconnect()
        }
    }

    @Composable
    private fun IptvScreen() {

        var loading by remember {
            mutableStateOf(true)
        }

        var errorMessage by remember {
            mutableStateOf("")
        }

        var channels by remember {
            mutableStateOf(
                emptyList<IptvChannel>()
            )
        }

        var selectedCategory by remember {
            mutableStateOf<String?>(null)
        }

        var searchText by remember {
            mutableStateOf("")
        }

        LaunchedEffect(Unit) {

            loadIptv(
                onSuccess = { loadedChannels ->
                    channels = loadedChannels
                    loading = false
                },
                onError = { message ->
                    errorMessage = message
                    loading = false
                }
            )
        }

        when {

            loading -> {
                LoadingScreen()
            }

            errorMessage.isNotBlank() -> {
                ErrorScreen(
                    message = errorMessage
                )
            }

            else -> {

                val categories =
                    buildCategoryList(channels)

                val normalizedSearch =
                    searchText.trim()

                val filteredChannels =
                    channels.filter { channel ->

                        val matchesCategory =
                            selectedCategory == null ||
                                    channelBelongsToCategory(
                                        channel.category,
                                        selectedCategory
                                    )

                        val matchesSearch =
                            normalizedSearch.isBlank() ||
                                    channel.name.contains(
                                        normalizedSearch,

    private fun loadIptv(
        onSuccess: (List<IptvChannel>) -> Unit,
        onError: (String) -> Unit
    ) {
        Thread {
            try {

                val response = postJson(
                    url = "$apiBaseUrl/api/iptv",
                    body = JSONObject().apply {
                        put("device_uuid", deviceUuid)
                        put("license_key", licenseKey)
                    }
                )

                if (!response.optBoolean("licensed", false)) {
                    throw IllegalStateException(
                        response.optString(
                            "message",
                            "IPTV eriÅŸimi yok."
                        )
                    )
                }

                val channelArray =
                    response.optJSONArray("channels")

                val loadedChannels =
                    mutableListOf<IptvChannel>()

                if (channelArray != null) {

                    for (index in 0 until channelArray.length()) {

                        val item =
                            channelArray.getJSONObject(index)

                        loadedChannels.add(
                            IptvChannel(
                                id = item.optInt("id"),

                                name = item.optString("name"),

                                streamUrl = item.optString(
                                    "stream_url"
                                ),

                                logoUrl = item
                                    .optString("logo_url")
                                    .ifBlank {
                                        null
                                    },

                                category = item
                                    .optString("category")
                                    .ifBlank {
                                        null
                                    }
                            )
                        )
                    }
                }

                runOnUiThread {
                    onSuccess(loadedChannels)
                }

            } catch (exception: Exception) {

                runOnUiThread {
                    onError(
                        exception.message
                            ?: "IPTV verileri alÄ±namadÄ±."
                    )
                }
            }
        }.start()
    }

    private fun postJson(
        url: String,
        body: JSONObject
    ): JSONObject {

        val connection =
            URL(url).openConnection() as HttpURLConnection

        try {

            connection.requestMethod = "POST"
            connection.connectTimeout = 10_000
            connection.readTimeout = 15_000
            connection.doOutput = true

            connection.setRequestProperty(
                "Content-Type",
                "application/json"
            )

            connection.setRequestProperty(
                "Accept",
                "application/json"
            )

            connection.outputStream.use { output ->
                output.write(
                    body.toString()
                        .toByteArray(
                            Charsets.UTF_8
                        )
                )
            }

            val responseCode =
                connection.responseCode

            val responseText =
                if (responseCode in 200..299) {
                    connection.inputStream
                        .bufferedReader()
                        .use {
                            it.readText()
                        }
                } else {
                    connection.errorStream
                        ?.bufferedReader()
                        ?.use {
                            it.readText()
                        }
                        ?: ""
                }

            if (responseCode !in 200..299) {

                throw IllegalStateException(
                    try {
                        JSONObject(responseText)
                            .optString(
                                "message",
                                "IPTV API hatasÄ±: $responseCode"
                            )
                    } catch (_: Exception) {
                        "IPTV API hatasÄ±: $responseCode"
                    }
                )
            }

            return JSONObject(responseText)

        } finally {
            connection.disconnect()
        }
    }

    private fun ChannelLogo(
        logoUrl: String?,

    ) {

        var bitmap by remember(logoUrl) {

        }

            bitmap = loadLogo(logoUrl)
        }

        Box(

                            .surfaceVariant
                    ),

                Alignment.Center
        ) {

            if (image != null) {

                Image(
                    bitmap =
                        image.asImageBitmap(),

                    contentDescription = null,

                    modifier =
                        Modifier.fillMaxSize(),

                    contentScale =
                        ContentScale.Fit
                )

            } else {

                Text(
                    text = "TV",

                    style =
                        MaterialTheme
                            .typography
                            .labelLarge
                )
            }
        }
    }

    private fun loadLogo(
        urlString: String
    ): Bitmap? {

        return try {

            val connection =
                URL(urlString)
                    .openConnection() as HttpURLConnection

            connection.connectTimeout = 5000
            connection.readTimeout = 5000

            connection.setRequestProperty(
                "User-Agent",
                "KaradagCarTV/1.0"
            )

            connection.doInput = true
            connection.connect()

            val bitmap =
                connection.inputStream.use {
                    BitmapFactory.decodeStream(it)
                }

            connection.disconnect()

            if (bitmap != null) {
                logoCache.put(
                    urlString,
                    bitmap
                )
            }

            bitmap

        } catch (_: Exception) {
            null
        }
    }

    private fun buildCategoryList(
        channels: List<IptvChannel>
    ): List<String> {

        val parts =
            channels
                .flatMap { channel ->

                    channel.category
                        ?.split(";")
                        ?: emptyList()
                }
                .map {
                    it.trim()
                }
                .filter {
                    it.isNotBlank()
                }
                .distinct()

        val preferredOrder =
            listOf(
                "News",
                "Sports",
                "Kids",
                "Music",
                "Movies",
                "Entertainment",
                "General",
                "Business",
                "Documentary",
                "Education",
                "Culture",
                "Travel",
                "Lifestyle",
                "Religious",
                "Animation",
                "Series",
                "Undefined"
            )

        return parts.sortedWith(
            compareBy(
                { value ->

                    val index =
                        preferredOrder.indexOfFirst {
                            it.equals(
                                value,
                                ignoreCase = true
                            )
                        }

                    if (index == -1) {
                        999
                    } else {
                        index
                    }
                },

                {
                    it.lowercase(
                        Locale.getDefault()
                    )
                }
            )
        )
    }

    }
